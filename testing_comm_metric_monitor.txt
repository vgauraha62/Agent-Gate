# 1. Build
zig build -Doptimize=ReleaseFast
# 2. Generate denials
curl -X POST http://localhost:8080/check -H "Content-Type: application/json" \
  -d '{"path":"/admin","method":"GET"}'
# 3. Query denials
curl "http://localhost:8080/denied-requests?limit=10"
# 4. Filter by agent (use 64-char hex string)
curl "http://localhost:8080/denied-requests?agent=abc123...&limit=50"
# 5. Filter by time (microseconds since epoch)
curl "http://localhost:8080/denied-requests?since=1750521600000000"
